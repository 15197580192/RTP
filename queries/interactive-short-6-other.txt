MATCH (m:Comment {id: "21990305681072" })
OPTIONAL MATCH (m)-[:replyOf]->(p:Comment)
RETURN CASE WHEN p IS NULL THEN m ELSE p END

MATCH (m:Comment)-[:replyOf]->(p:Post)
RETURN p

MATCH (p:Post)<-[:containerOf]-(f:Forum)
RETURN f

MATCH (f:Forum)-[:hasModerator]->(mod:Person)
RETURN
    f.id AS forumId,
    f.title AS forumTitle,
    mod.id AS moderatorId,
    mod.firstName AS moderatorFirstName,
    mod.lastName AS moderatorLastName
